82static void apply_mode(const char *mode_name, const char *direction)
83{
84 if (msg_constants::msg_label != NULL)
85 {
86 lv_obj_del(msg_constants::msg_label);
87 msg_constants::msg_label = NULL;
88 }
89
90 msg_constants::msg_label = lv_label_create(lv_scr_act());

92 snprintf(buffer, sizeof(buffer), "Applied %s Mode: %s.", mode_name, direction);

94
95 other_constants::program.push_back(direction);
96
97 lv_obj_set_style_text_font(msg_constants::msg_label, &lv_font_montserrat_24, 0);
98 lv_obj_align(msg_constants::msg_label, LV_ALIGN_BOTTOM_MID, 0, -20);
99 msg_constants::msg_start_time = millis();
100}

150// ======== MAIN BUTTON CALLBACK ========
166static void main_btnm_event_cb(lv_event_t *e)
167{
168 lv_obj_t *btnm = (lv_obj_t *)lv_event_get_target(e);
169 const char *dir_txt = lv_btnmatrix_get_btn_text(
170 btnm, lv_btnmatrix_get_selected_btn(btnm));
171
172 if (dir_txt == NULL)
173 return;
174
175 if (strcmp(dir_txt, "Forward") == 0 || strcmp(dir_txt, "Backward") == 0)
176 {
177 apply_mode("Straight", dir_txt);
178 }
179 else if (strcmp(dir_txt, "Left") == 0 || strcmp(dir_txt, "Right") == 0)
180 {
181 apply_mode("Sideways", dir_txt);
182 }
183 else if (strcmp(dir_txt, "45") == 0 || strcmp(dir_txt, "135") == 0 || strcmp(dir_txt, "225") == 0 || strcmp(dir_txt, "315") == 0)
184 {
185 apply_mode("Diagonal", dir_txt);
186 }
187 else if (strcmp(dir_txt, "Right Fwd") == 0 || strcmp(dir_txt, "Right Bwd") == 0 || strcmp(dir_txt, "Left Fwd") == 0 || strcmp(dir_txt, "Left Bwd") == 0)
188 {
189 apply_mode("Pivot", dir_txt);
190 }
191 else if (strcmp(dir_txt, "Front Right") == 0 || strcmp(dir_txt, "Front Left") == 0 || strcmp(dir_txt, "Rear Right") == 0 || strcmp(dir_txt, "Rear Left") == 0)
192 {
193 apply_mode("Pivot Sideways", dir_txt);
194 }
195 else if (strcmp(dir_txt, "CCW") == 0 || strcmp(dir_txt, "CW") == 0)
196 {
197 apply_mode("Rotate", dir_txt);
198 }
199 else if (strcmp(dir_txt, "Prgm Car") == 0)
200 {
201 btn_Run_Program_event_cb(e);
202 }
203}

266static void toggle_fwdrev_cb(lv_event_t *e)
267{
268 lv_obj_t *btn = (lv_obj_t *)lv_event_get_target(e);

270

275 lv_label_set_text(label, "Rev");
276 motor_constants::motor_direction = 1;
277 }
278 else
279 {
280 lv_label_set_text(label, "Fwd");
281 motor_constants::motor_direction = -1;
282 }
283
284 Serial.print("Mode: ");
285 Serial.println(lv_label_get_text(label));
286 Serial.print("Direction set to ");
287 Serial.println(motor_constants::motor_direction);

385// ===== Speed Sliders =====
386void create_speed_sliders(lv_obj_t *parent)
387{
388 const char *labels[] = {"FR", "FL", "RR", "RL"};
389 int y_positions[] = {20, 60, 100, 140};
390
391 for (int i = 0; i < 4; i++)
392 {
393 // Create label
394 lv_obj_t *label = lv_label_create(parent);
395 lv_label_set_text(label, labels[i]);
396 lv_obj_align(label, LV_ALIGN_TOP_RIGHT, -230, y_positions[i]);
397
398 // Create slider
399 lv_obj_t *slider = lv_slider_create(parent);
400 lv_slider_set_range(slider, 0, 255);
401 lv_obj_set_size(slider, 150, 20);
402 lv_obj_align_to(slider, label, LV_ALIGN_OUT_RIGHT_MID, 15, 0);
403
404 // Create value label
405 lv_obj_t *value_label = lv_label_create(parent);
406 char buf[4];
407 snprintf(buf, sizeof(buf), "%d", lv_slider_get_value(slider));
408 lv_label_set_text(value_label, buf);
409 lv_obj_align_to(value_label, slider, LV_ALIGN_OUT_RIGHT_MID, 15, 0);
410
411 // Add event callback that updates the value label
412 lv_obj_add_event_cb(
413 slider, [](lv_event_t *e)
414 {
415 lv_obj_t* s = (lv_obj_t*)lv_event_get_target(e);
416 lv_obj_t* val_lbl = (lv_obj_t*)lv_event_get_user_data(e);
417 char buf[4]; // Buffer to hold the string representation of the slider value (3 digits + null terminator)
418 snprintf(buf, sizeof(buf), "%d", lv_slider_get_value(s));
419 lv_label_set_text(val_lbl, buf); },
420 LV_EVENT_VALUE_CHANGED, value_label);
421
422 // Add slider event callback to update speed
423 lv_obj_add_event_cb(slider, slider_event_cb, LV_EVENT_VALUE_CHANGED, (void *)labels[i]);
424 }
425}

459void FR_move(int speed)
460{
461 if (speed >= 0)
462 {
463 digitalWrite(MotorPins::IN1_B, HIGH);
464 digitalWrite(MotorPins::IN2_B, LOW);
465 analogWrite(MotorPins::ENA_B, speed);
466 }
467 else
468 {
469 digitalWrite(MotorPins::IN1_B, LOW);
470 digitalWrite(MotorPins::IN2_B, HIGH);
471 analogWrite(MotorPins::ENA_B, -speed);
472 }
