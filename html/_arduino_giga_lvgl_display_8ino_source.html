84static void apply_mode(const char *mode_name, const char *direction)
85{
86 if (msg_constants::msg_label != NULL)
87 {
88 lv_obj_del(msg_constants::msg_label);
89 msg_constants::msg_label = NULL;
90 }
91
92 msg_constants::msg_label = lv_label_create(lv_scr_act());
93 char buffer[64];
94 snprintf(buffer, sizeof(buffer), "Applied %s Mode: %s.", mode_name, direction);
95 lv_label_set_text(msg_constants::msg_label, buffer);
96
97 other_constants::program.push_back(direction);
98
99 lv_obj_set_style_text_font(msg_constants::msg_label, &lv_font_montserrat_24, 0);
100 lv_obj_align(msg_constants::msg_label, LV_ALIGN_BOTTOM_MID, 0, -20);
101 msg_constants::msg_start_time = millis();
102}

155// ======== MAIN BUTTON CALLBACK ========
171static void main_btnm_event_cb(lv_event_t *e)
172{
173 lv_obj_t *btnm = (lv_obj_t *)lv_event_get_target(e);
174 const char *dir_txt = lv_btnmatrix_get_btn_text(
175 btnm, lv_btnmatrix_get_selected_btn(btnm));
176
177 if (dir_txt == NULL)
178 return;
179
180 if (strcmp(dir_txt, "Forward") == 0 || strcmp(dir_txt, "Backward") == 0)
181 {
182 apply_mode("Straight", dir_txt);
183 }
184 else if (strcmp(dir_txt, "Left") == 0 || strcmp(dir_txt, "Right") == 0)
185 {
186 apply_mode("Sideways", dir_txt);
187 }
188 else if (strcmp(dir_txt, "45") == 0 || strcmp(dir_txt, "135") == 0 || strcmp(dir_txt, "225") == 0 || strcmp(dir_txt, "315") == 0)
189 {
190 apply_mode("Diagonal", dir_txt);
191 }
192 else if (strcmp(dir_txt, "Right Fwd") == 0 || strcmp(dir_txt, "Right Bwd") == 0 || strcmp(dir_txt, "Left Fwd") == 0 || strcmp(dir_txt, "Left Bwd") == 0)
193 {
194 apply_mode("Pivot", dir_txt);
195 }
196 else if (strcmp(dir_txt, "Front Right") == 0 || strcmp(dir_txt, "Front Left") == 0 || strcmp(dir_txt, "Rear Right") == 0 || strcmp(dir_txt, "Rear Left") == 0)
197 {
198 apply_mode("Pivot Sideways", dir_txt);
199 }
200 else if (strcmp(dir_txt, "CCW") == 0 || strcmp(dir_txt, "CW") == 0)
201 {
202 apply_mode("Rotate", dir_txt);
203 }
204 else if (strcmp(dir_txt, "Prgm Car") == 0)
205 {
206 btn_Run_Program_event_cb(e);
207 }
208}

271static void toggle_fwdrev_cb(lv_event_t *e)
272{
273 lv_obj_t *btn = (lv_obj_t *)lv_event_get_target(e);
274 lv_obj_t *label = lv_obj_get_child(btn, 0);
275
276 const char *txt = lv_label_get_text(label);
277 // If txt is "Fwd", change to "Rev" and assign motor_direction 1 else change to "Fwd" and assign motor_direction -1
278 if (strcmp(txt, "Fwd") == 0)
279 {
280 lv_label_set_text(label, "Rev");
281 motor_constants::motor_direction = 1;
282 }
283 else
284 {
285 lv_label_set_text(label, "Fwd");
286 motor_constants::motor_direction = -1;
287 }
288
289 Serial.print("Mode: ");
290 Serial.println(lv_label_get_text(label));
291 Serial.print("Direction set to ");
292 Serial.println(motor_constants::motor_direction);
293}

390// ===== Speed Sliders =====
391void create_speed_sliders(lv_obj_t *parent)
392{
393 const char *labels[] = {"FR", "FL", "RR", "RL"};
394 int y_positions[] = {20, 60, 100, 140};
395
396 for (int i = 0; i < 4; i++)
397 {
398 // Create label
399 lv_obj_t *label = lv_label_create(parent);
400 lv_label_set_text(label, labels[i]);
401 lv_obj_align(label, LV_ALIGN_TOP_RIGHT, -230, y_positions[i]);
402
403 // Create slider
404 lv_obj_t *slider = lv_slider_create(parent);
405 lv_slider_set_range(slider, 0, 255);
406 lv_obj_set_size(slider, 150, 20);
407 lv_obj_align_to(slider, label, LV_ALIGN_OUT_RIGHT_MID, 15, 0);
408
409 // Create value label
410 lv_obj_t *value_label = lv_label_create(parent);
411 char buf[4];
412 snprintf(buf, sizeof(buf), "%d", lv_slider_get_value(slider));
413 lv_label_set_text(value_label, buf);
414 lv_obj_align_to(value_label, slider, LV_ALIGN_OUT_RIGHT_MID, 15, 0);
415
416 // Add event callback that updates the value label
417 lv_obj_add_event_cb(
418 slider, [](lv_event_t *e)
419 {
420 lv_obj_t* s = (lv_obj_t*)lv_event_get_target(e);
421 lv_obj_t* val_lbl = (lv_obj_t*)lv_event_get_user_data(e);
422 char buf[4]; // Buffer to hold the string representation of the slider value (3 digits + null terminator)
423 snprintf(buf, sizeof(buf), "%d", lv_slider_get_value(s));
424 lv_label_set_text(val_lbl, buf); },
425 LV_EVENT_VALUE_CHANGED, value_label);
426
427 // Add slider event callback to update speed
428 lv_obj_add_event_cb(slider, slider_event_cb, LV_EVENT_VALUE_CHANGED, (void *)labels[i]);
429 }
430}

464void FR_move(int speed)
465{
466 if (speed >= 0)

468 digitalWrite(MotorPins::IN1_B, HIGH);
469 digitalWrite(MotorPins::IN2_B, LOW);

474 digitalWrite(MotorPins::IN1_B, LOW);
