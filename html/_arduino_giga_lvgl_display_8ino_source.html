133// ======== MAIN BUTTON CALLBACK ========
149static void main_btnm_event_cb(lv_event_t *e)
150{
151 lv_obj_t *btnm = (lv_obj_t *)lv_event_get_target(e);
152 const char *txt = lv_btnmatrix_get_btn_text(
153 btnm, lv_btnmatrix_get_selected_btn(btnm));
154
155 if (txt == NULL)
156 return;
157
158 if (strcmp(txt, "Forward") == 0 || strcmp(txt, "Backward") == 0)
159 {
160 apply_mode("Straight", txt);
161 }
162 else if (strcmp(txt, "Left") == 0 || strcmp(txt, "Right") == 0)
163 {
164 apply_mode("Sideways", txt);
165 }
166 else if (strcmp(txt, "45") == 0 || strcmp(txt, "135") == 0 || strcmp(txt, "225") == 0 || strcmp(txt, "315") == 0)
167 {
168 apply_mode("Diagonal", txt);
169 }
170 else if (strcmp(txt, "Right Fwd") == 0 || strcmp(txt, "Right Bwd") == 0 || strcmp(txt, "Left Fwd") == 0 || strcmp(txt, "Left Bwd") == 0)
171 {
172 apply_mode("Pivot", txt);
173 }
174 else if (strcmp(txt, "Front Right") == 0 || strcmp(txt, "Front Left") == 0 || strcmp(txt, "Rear Right") == 0 || strcmp(txt, "Rear Left") == 0)
175 {
176 apply_mode("Pivot Sideways", txt);
177 }
178 else if (strcmp(txt, "CCW") == 0 || strcmp(txt, "CW") == 0)
179 {
180 apply_mode("Rotate", txt);
181 }
182 else if (strcmp(txt, "Prgm Car") == 0)
183 {
184 btn_Run_Program_event_cb(e);
185 }
186}

363// ===== Speed Sliders =====
364void create_speed_sliders(lv_obj_t *parent)
365{
366 const char *labels[] = {"FR", "FL", "RR", "RL"};
367 int y_positions[] = {20, 60, 100, 140};
368
369 for (int i = 0; i < 4; i++)
370 {
371 // Create label
372 lv_obj_t *label = lv_label_create(parent);
373 lv_label_set_text(label, labels[i]);
374 lv_obj_align(label, LV_ALIGN_TOP_RIGHT, -230, y_positions[i]);
375
376 // Create slider
377 lv_obj_t *slider = lv_slider_create(parent);
378 lv_slider_set_range(slider, 0, 255);
379 lv_obj_set_size(slider, 150, 20);
380 lv_obj_align_to(slider, label, LV_ALIGN_OUT_RIGHT_MID, 15, 0);
381
382 // Create value label
383 lv_obj_t *value_label = lv_label_create(parent);
384 char buf[4];
385 snprintf(buf, sizeof(buf), "%d", lv_slider_get_value(slider));
386 lv_label_set_text(value_label, buf);
387 lv_obj_align_to(value_label, slider, LV_ALIGN_OUT_RIGHT_MID, 15, 0);
388
389 // Add event callback that updates the value label
390 lv_obj_add_event_cb(
391 slider, [](lv_event_t *e)
392 {
393 lv_obj_t* s = (lv_obj_t*)lv_event_get_target(e);
394 lv_obj_t* val_lbl = (lv_obj_t*)lv_event_get_user_data(e);
395 char buf[4]; // Buffer to hold the string representation of the slider value (3 digits + null terminator)
396 snprintf(buf, sizeof(buf), "%d", lv_slider_get_value(s));
397 lv_label_set_text(val_lbl, buf); },
398 LV_EVENT_VALUE_CHANGED, value_label);
399
400 // Add slider event callback to update speed
401 lv_obj_add_event_cb(slider, slider_event_cb, LV_EVENT_VALUE_CHANGED, (void *)labels[i]);
402 }
403}
