70static void apply_mode(const char *mode_name, const char *direction)
71{
72 if (msg_label != NULL)
73 {
74 lv_obj_del(msg_label);
75 msg_label = NULL;
76 }
77
78 msg_label = lv_label_create(lv_scr_act());
79
80 char buffer[64];
81 snprintf(buffer, sizeof(buffer), "Applied %s Mode: %s.", mode_name, direction);
82 lv_label_set_text(msg_label, buffer);
83
84 program.push_back(direction);
85
86 lv_obj_set_style_text_font(msg_label, &lv_font_montserrat_24, 0);
87 lv_obj_align(msg_label, LV_ALIGN_BOTTOM_MID, 0, -20);
88 msg_start_time = millis();
89}

140// ======== MAIN BUTTON CALLBACK ========
156static void main_btnm_event_cb(lv_event_t *e)
157{
158 lv_obj_t *btnm = (lv_obj_t *)lv_event_get_target(e);
159 const char *txt = lv_btnmatrix_get_btn_text(
160 btnm, lv_btnmatrix_get_selected_btn(btnm));
161
162 if (txt == NULL)
163 return;
164
165 if (strcmp(txt, "Forward") == 0 || strcmp(txt, "Backward") == 0)
166 {
167 apply_mode("Straight", txt);
168 }
169 else if (strcmp(txt, "Left") == 0 || strcmp(txt, "Right") == 0)
170 {
171 apply_mode("Sideways", txt);
172 }
173 else if (strcmp(txt, "45") == 0 || strcmp(txt, "135") == 0 || strcmp(txt, "225") == 0 || strcmp(txt, "315") == 0)
174 {
175 apply_mode("Diagonal", txt);
176 }
177 else if (strcmp(txt, "Right Fwd") == 0 || strcmp(txt, "Right Bwd") == 0 || strcmp(txt, "Left Fwd") == 0 || strcmp(txt, "Left Bwd") == 0)
178 {
179 apply_mode("Pivot", txt);
180 }
181 else if (strcmp(txt, "Front Right") == 0 || strcmp(txt, "Front Left") == 0 || strcmp(txt, "Rear Right") == 0 || strcmp(txt, "Rear Left") == 0)
182 {
183 apply_mode("Pivot Sideways", txt);
184 }
185 else if (strcmp(txt, "CCW") == 0 || strcmp(txt, "CW") == 0)
186 {
187 apply_mode("Rotate", txt);
188 }
189 else if (strcmp(txt, "Prgm Car") == 0)
190 {
191 btn_Run_Program_event_cb(e);
192 }
193}

370// ===== Speed Sliders =====
371void create_speed_sliders(lv_obj_t *parent)
372{
373 const char *labels[] = {"FR", "FL", "RR", "RL"};
374 int y_positions[] = {20, 60, 100, 140};
375
376 for (int i = 0; i < 4; i++)
377 {
378 // Create label
379 lv_obj_t *label = lv_label_create(parent);
380 lv_label_set_text(label, labels[i]);
381 lv_obj_align(label, LV_ALIGN_TOP_RIGHT, -230, y_positions[i]);
382
383 // Create slider
384 lv_obj_t *slider = lv_slider_create(parent);
385 lv_slider_set_range(slider, 0, 255);
386 lv_obj_set_size(slider, 150, 20);
387 lv_obj_align_to(slider, label, LV_ALIGN_OUT_RIGHT_MID, 15, 0);
388
389 // Create value label
390 lv_obj_t *value_label = lv_label_create(parent);
391 char buf[4];
392 snprintf(buf, sizeof(buf), "%d", lv_slider_get_value(slider));
393 lv_label_set_text(value_label, buf);
394 lv_obj_align_to(value_label, slider, LV_ALIGN_OUT_RIGHT_MID, 15, 0);
395
396 // Add event callback that updates the value label
397 lv_obj_add_event_cb(
398 slider, [](lv_event_t *e)
399 {
400 lv_obj_t* s = (lv_obj_t*)lv_event_get_target(e);
401 lv_obj_t* val_lbl = (lv_obj_t*)lv_event_get_user_data(e);
402 char buf[4]; // Buffer to hold the string representation of the slider value (3 digits + null terminator)
403 snprintf(buf, sizeof(buf), "%d", lv_slider_get_value(s));
404 lv_label_set_text(val_lbl, buf); },
405 LV_EVENT_VALUE_CHANGED, value_label);
406
407 // Add slider event callback to update speed
408 lv_obj_add_event_cb(slider, slider_event_cb, LV_EVENT_VALUE_CHANGED, (void *)labels[i]);
409 }
410}
